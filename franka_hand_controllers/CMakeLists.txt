cmake_minimum_required(VERSION 3.16)
project(franka_hand_controllers)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(joint_trajectory_controller REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(controller_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(yaml_cpp_vendor REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(moveit_servo REQUIRED)
find_package(moveit_core REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(sensor_msgs REQUIRED)
add_executable(feedback_servo src/feedback_servo.cpp)
ament_target_dependencies(feedback_servo moveit_core)
target_link_libraries(feedback_servo moveit_servo::moveit_servo_lib_cpp)
install(TARGETS feedback_servo DESTINATION lib/${PROJECT_NAME})
add_library(${PROJECT_NAME} SHARED src/robot_clock_controller.cpp src/rolling_controller.cpp src/clocked_mock.cpp)
set_target_properties(${PROJECT_NAME} PROPERTIES UNITY_BUILD ON)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
  $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME}
  joint_trajectory_controller hardware_interface controller_interface pluginlib rclcpp std_msgs std_srvs sensor_msgs)
target_link_libraries(${PROJECT_NAME} hardware_interface::mock_components yaml-cpp)
target_link_libraries(feedback_servo ${PROJECT_NAME})
pluginlib_export_plugin_description_file(controller_interface controller_plugins.xml)
pluginlib_export_plugin_description_file(hardware_interface mock_plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT ${PROJECT_NAME}Targets LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(${PROJECT_NAME}Targets HAS_LIBRARY_TARGET)
ament_export_dependencies(moveit_core yaml_cpp_vendor joint_trajectory_controller hardware_interface controller_interface pluginlib rclcpp std_msgs std_srvs sensor_msgs)
ament_package()
